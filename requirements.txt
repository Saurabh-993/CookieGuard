# ============================================================
# CookieGuard - Python dependencies
#
# Install with uv (fast):   uv pip install -r requirements.txt
#
# NOTE ON VERSION RANGES
# ----------------------
# We use `>=` (minimum version) rather than `==` (exact pin) here.
# Reason: an exact pin can point at a version that has no pre-built
# "wheel" for your Python version, forcing pip/uv to COMPILE the
# package from source — which needs a Rust or C compiler and usually
# fails on Windows. Ranges let the resolver pick a version that has
# a ready-made wheel for your interpreter.
#
# For reproducible builds we lock exact versions separately:
#     uv pip freeze > requirements.lock.txt
# See docs/TEACHING.md §25 for the full explanation.
# ============================================================

# --- Phase 1: Scanner ---
# Playwright drives a REAL browser (Chromium) from Python so we can
# see every cookie a site actually sets, including ones created by
# JavaScript after the page loads.
#
# ⚠ IF YOU RUN LOCALLY *AND* IN DOCKER, KEEP THESE ALIGNED.
# Playwright is the one dependency coupled to something outside Python: the
# browser binaries on disk. The Docker image pins it to match its base image
# (see `ARG PLAYWRIGHT_VERSION` in the Dockerfile), so a floating range here
# is fine for the container — the Dockerfile constrains it.
#
# Locally, `playwright install chromium` downloads whatever your installed
# package expects, so local stays consistent by itself. The failure only
# happens when a floating package meets a fixed browser directory, which is
# exactly what a base image is.
playwright>=1.48

# tldextract gives us Mozilla's Public Suffix List, which is the only correct
# way to reduce "www.bbc.co.uk" to "bbc.co.uk" rather than "co.uk".
# We pin it to the bundled offline snapshot in scanner/domains.py so it never
# makes a network call at import time - see docs/TEACHING.md §43.
tldextract>=5.1

# --- Phase 3: REST API (installed now so the env is complete) ---
# FastAPI = the web framework that will expose our scan endpoints.
fastapi>=0.115

# Uvicorn = the ASGI web server that actually runs FastAPI.
# (FastAPI is the application; Uvicorn is the engine that serves it.)
uvicorn[standard]>=0.30

# Pydantic = data validation. FastAPI uses it to check that incoming
# JSON has the right shape before our code ever sees it.
pydantic>=2.9

# --- Testing (used by GitHub Actions CI in Phase 6) ---
pytest>=8.3

# FastAPI's TestClient needs an HTTP client library to drive the app in-process.
# Which one depends on your Starlette version:
#   Starlette < 0.49  ->  httpx
#   Starlette >= 0.49 ->  httpx2   (it raises a clear error telling you which)
# Installing both is harmless and means the tests run on either version.
httpx>=0.27
httpx2>=2.0
